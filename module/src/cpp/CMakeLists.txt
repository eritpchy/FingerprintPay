cmake_minimum_required(VERSION 3.22.1)
project(${MODULE_NAME})
set(CMAKE_CXX_STANDARD 17)
add_definitions(-DMODULE_NAME="${MODULE_NAME}")
include_directories("${FINGERPRINT_ROOT}/module/vendor")
set(PAYLOAD "${FINGERPRINT_ROOT}/module/build/payload/payload.bin")
set(RESOURCE_PACKAGE "${FINGERPRINT_ROOT}/module/build/payload/resources.apk")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${RESOURCE_PACKAGE}")
file(SHA256 "${RESOURCE_PACKAGE}" FP_RESOURCE_HASH)
add_definitions(-DFP_RESOURCE_SHA256="${FP_RESOURCE_HASH}")
file(TO_CMAKE_PATH "${PAYLOAD}" PAYLOAD)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/payload.S" ".section .rodata\n.balign 8\n.global fp_payload_start\n.hidden fp_payload_start\nfp_payload_start:\n.incbin \"${PAYLOAD}\"\n.global fp_payload_end\n.hidden fp_payload_end\nfp_payload_end:\n.section .note.GNU-stack,\"\",%progbits\n")
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/payload.S" PROPERTIES OBJECT_DEPENDS "${PAYLOAD}")
enable_language(ASM)
add_library(${MODULE_NAME} SHARED fingerprint.cpp art_symbol.cpp memory_dex.cpp debug_payload.cpp zygisk_main.cpp "${CMAKE_CURRENT_BINARY_DIR}/payload.S")
target_compile_definitions(${MODULE_NAME} PRIVATE "$<$<CONFIG:Debug>:FP_DEBUG_FILE_DEX=1>")
target_compile_options(${MODULE_NAME} PRIVATE -Oz -fvisibility=hidden -ffunction-sections -fdata-sections)
set_source_files_properties(fingerprint.cpp art_symbol.cpp memory_dex.cpp debug_payload.cpp zygisk_main.cpp PROPERTIES COMPILE_OPTIONS "-fno-exceptions;-fno-rtti")
target_link_options(${MODULE_NAME} PRIVATE -Wl,--gc-sections -Wl,--hash-style=both)
target_link_libraries(${MODULE_NAME} log dl)
set(SODIUM "${FINGERPRINT_ROOT}/module/build/sodium/${ANDROID_ABI}")
add_executable(fpconfig config_daemon.cpp)
target_include_directories(fpconfig PRIVATE "${SODIUM}/include")
target_link_libraries(fpconfig "${SODIUM}/lib/libsodium.a")
set_target_properties(fpconfig PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${FINGERPRINT_ROOT}/module/build/bin/${ANDROID_ABI}")

target_compile_options(fpconfig PRIVATE -Os -ffunction-sections -fdata-sections)
target_link_options(fpconfig PRIVATE -Wl,--gc-sections)
add_custom_command(TARGET fpconfig POST_BUILD COMMAND ${CMAKE_STRIP} --strip-unneeded $<TARGET_FILE:fpconfig>)
